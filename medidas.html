Resumo Dashboard DAX

Documentação da medida Resumo Dashboard.

Fórmula DAX

VAR TotalValor = [09 - Vsenhas]
VAR TotalVolume = [10 - Volume senhas]
VAR TicketMedio = [16 - ticket médio]

VAR BandeiraTop =
TOPN(
    1,
    ALL('DimClientes'[BANDEIRA]),
    [09 - Vsenhas],
    DESC
)

VAR NomeBandeira =
MAXX(BandeiraTop, 'DimClientes'[BANDEIRA])

VAR PercBandeira =
MAXX(BandeiraTop, [%GeralV])

VAR RetornoBandeiraTop =
    CALCULATE(
        [09 - Vsenhas],
        BandeiraTop
    )

VAR FaturamentoBandeiraTop =
    CALCULATE(
        [01 - Faturamento],
        BandeiraTop
    )

VAR PercSobreFaturamento =
    DIVIDE(RetornoBandeiraTop, FaturamentoBandeiraTop)

RETURN
"Foram registradas "
& FORMAT(TotalVolume,"#,##0")
& " senhas, totalizando "
& FORMAT(TotalValor,"R$ #,##0")
& " em retornos. Ticket médio de "
& FORMAT(TicketMedio,"R$ #,##0")
& ". A bandeira com maior impacto é "
& NomeBandeira
& ", representando "
& FORMAT(PercBandeira,"0.00%")
& " do total, e seu retorno equivale a "
& FORMAT(PercSobreFaturamento,"0.00%")
& " do faturamento próprio."

Insight_Retorno DAX

Documentação da medida Insight_Retorno.

Fórmula DAX

VAR TabelaBase =
    ADDCOLUMNS(
        ALLSELECTED('PEX 1031'[CAUSA_AGRUPADA]),
        "Valor", [03 - Valor Retorno]
    )

VAR Total =
    SUMX(TabelaBase, [Valor])

VAR TabelaRank =
    ADDCOLUMNS(
        TabelaBase,
        "Rank",
            RANKX(TabelaBase, [Valor], , DESC),
        "Acum",
            VAR RankAtual = RANKX(TabelaBase, [Valor], , DESC)
            RETURN
                DIVIDE(
                    SUMX(
                        FILTER(TabelaBase, RANKX(TabelaBase, [Valor], , DESC) <= RankAtual),
                        [Valor]
                    ),
                    Total
                )
    )

VAR Top1 =
    TOPN(1, TabelaBase, [Valor], DESC)

VAR NomeTop =
    MAXX(Top1, 'PEX 1031'[CAUSA_AGRUPADA])

VAR ValorTop =
    MAXX(Top1, [Valor])

VAR Ate80 =
    FILTER(TabelaRank, [Acum] <= 0.8)

VAR Qt =
    COUNTROWS(Ate80)

VAR Perc =
    MAXX(Ate80, [Acum])

RETURN
FORMAT(Perc, "0%") &
" do retorno vem de " &
Qt &
" causas, lideradas por " &
NomeTop &
" (" & FORMAT(ValorTop, "R$ #,##0") & ")."

Observações

Preencher regra de negócio e observações.
